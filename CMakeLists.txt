cmake_minimum_required(VERSION 3.22)
project(ConVERT VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(MSVC)
 set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
enable_testing()
add_executable(RulesTests Tests/RulesTests.cpp)
add_test(NAME ConversionRules COMMAND RulesTests)
option(CONVERT_RULES_ONLY "Build pure URL validation tests without JUCE" OFF)
if(CONVERT_RULES_ONLY)
 return()
endif()
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.6 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
juce_add_plugin(ConVERT COMPANY_NAME "SosaWest" BUNDLE_ID "com.sosawest.convert"
 PLUGIN_MANUFACTURER_CODE Sosw PLUGIN_CODE Cv27 PRODUCT_NAME "ConVERT"
 FORMATS VST3 Standalone IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE
 EDITOR_WANTS_KEYBOARD_FOCUS TRUE COPY_PLUGIN_AFTER_BUILD FALSE VST3_CATEGORIES Fx Tools)
target_sources(ConVERT PRIVATE Source/Plugin.cpp)
target_compile_definitions(ConVERT PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(ConVERT PRIVATE juce::juce_audio_utils PUBLIC juce::juce_recommended_config_flags)
option(CONVERT_NATIVE_TESTS "Build processor/state/engine integration tests" OFF)
if(CONVERT_NATIVE_TESTS)
 add_executable(FakeDownloader Tests/FakeDownloader.cpp)
 add_executable(NativeSmoke Tests/NativeSmoke.cpp)
 target_link_libraries(NativeSmoke PRIVATE ConVERT)
 target_include_directories(NativeSmoke PRIVATE "${juce_SOURCE_DIR}/modules")
 target_compile_definitions(NativeSmoke PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1 JUCE_USE_CURL=0 JUCE_WEB_BROWSER=0)
 add_test(NAME NativeSmoke COMMAND NativeSmoke $<TARGET_FILE:FakeDownloader>)
endif()
